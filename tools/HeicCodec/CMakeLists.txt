cmake_minimum_required(VERSION 3.24)
project(StarshotHeic LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_PLUGIN_LOADING OFF CACHE BOOL "" FORCE)
set(WITH_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_DOCUMENTATION OFF CACHE BOOL "" FORCE)
set(WITH_GDK_PIXBUF OFF CACHE BOOL "" FORCE)
set(WITH_X264 OFF CACHE BOOL "" FORCE)
set(WITH_OpenH264_DECODER OFF CACHE BOOL "" FORCE)
set(WITH_AOM_DECODER OFF CACHE BOOL "" FORCE)
set(WITH_AOM_ENCODER OFF CACHE BOOL "" FORCE)
set(WITH_LIBSHARPYUV OFF CACHE BOOL "" FORCE)
add_subdirectory("${STARSHOT_HEIF_SOURCE}" heif)
target_compile_definitions(heif PRIVATE LIBDE265_STATIC_BUILD)
target_compile_options(heif PRIVATE /utf-8)
add_executable(StarshotHeic main.cpp)
target_link_libraries(StarshotHeic PRIVATE heif)
target_include_directories(StarshotHeic PRIVATE "${STARSHOT_HEIF_SOURCE}/libheif/api" "${CMAKE_CURRENT_BINARY_DIR}/heif")
target_compile_options(StarshotHeic PRIVATE /utf-8)
